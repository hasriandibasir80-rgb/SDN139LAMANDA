// FILE FIX: js/modules/arsip-upload.js // FIX KASUS 1: Pisah storage key, dan render terbaru hanya dari upload import { db, auth } from '../firebase-config.js'; const KEY_UPLOAD = 'sdn139_upload_dokumen'; // KHUSUS UPLOAD const KEY_SIMPAN = 'sdn139_simpan_file'; // untuk referensi, jangan dipakai disini const form = document.getElementById('formUpload'); const fileInput = document.getElementById('fileInput'); const recentList = document.getElementById('recentList'); const btnText = document.getElementById('btnText'); const uploadStatus = document.getElementById('uploadStatus'); const dropZone = document.getElementById('dropZone'); const fileInfo = document.getElementById('fileInfo'); function getData() { return JSON.parse(localStorage.getItem(KEY_UPLOAD) || '[]'); } function saveData(data) { localStorage.setItem(KEY_UPLOAD, JSON.stringify(data)); } function dataURLtoBlob(dataURL) { const [meta, base64] = dataURL.split(','); const mime = meta.match(/:(.*?);/)[1]; const binary = atob(base64); const arr = new Uint8Array(binary.length); for (let i = 0; i < binary.length; i++) arr[i] = binary.charCodeAt(i); return new Blob([arr], { type: mime }); } function renderRecent() { const data = getData(); if (data.length === 0) { recentList.innerHTML = `

Belum ada dokumen yang diupload.

`; return; } recentList.innerHTML = data.slice(0, 5).map(item => `
${item.nama}
📅 ${item.tanggal} • 📁 ${item.kategori} • ${item.levelAkses}
PUBLIK
`).join(''); } // Drag & Drop UI dropZone?.addEventListener('click', () => fileInput.click()); dropZone?.addEventListener('dragover', e => { e.preventDefault(); dropZone.classList.add('dragover'); }); dropZone?.addEventListener('dragleave', () => dropZone.classList.remove('dragover')); dropZone?.addEventListener('drop', e => { e.preventDefault(); dropZone.classList.remove('dragover'); if (e.dataTransfer.files[0]) { fileInput.files = e.dataTransfer.files; showFileInfo(fileInput.files[0]); } }); fileInput?.addEventListener('change', () => { if (fileInput.files[0]) showFileInfo(fileInput.files[0]); }); function showFileInfo(file) { fileInfo.style.display = 'block'; fileInfo.innerHTML = `📄 ${file.name} (${(file.size/1024/1024).toFixed(2)} MB)`; } form?.addEventListener('submit', async (e) => { e.preventDefault(); const file = fileInput.files[0]; if (!file) return alert('Pilih file!'); if (file.size > 10 * 1024 * 1024) return alert('Max 10MB'); btnText.textContent = 'Mengupload...'; form.querySelector('button').disabled = true; const reader = new FileReader(); reader.onload = (ev) => { const data = getData(); data.unshift({ id: Date.now(), nama: document.getElementById('namaDokumen').value.trim(), kategori: document.getElementById('kategori').value, levelAkses: document.getElementById('levelAkses').value, deskripsi: document.getElementById('deskripsi').value, fileNameAsli: file.name, fileType: file.type, fileSize: file.size, base64: ev.target.result, tanggal: new Date().toLocaleDateString('id-ID', { day:'2-digit', month:'2-digit', year:'numeric' }), timestamp: Date.now(), sumber: 'upload-dokumen' }); saveData(data); renderRecent(); uploadStatus.innerHTML = `
✅ Dokumen "${file.name}" berhasil disimpan!
`; form.reset(); fileInfo.style.display = 'none'; btnText.textContent = '💾 Upload & Simpan Metadata'; form.querySelector('button').disabled = false; }; reader.onerror = () => { alert('Gagal baca file'); btnText.textContent = '💾 Upload & Simpan Metadata'; form.querySelector('button').disabled = false; }; reader.readAsDataURL(file); }); renderRecent(); // Expose global untuk katalog jika butuh window.lihatFileUpload = (id) => { const item = getData().find(f => f.id == id); if (!item) return; const blob = dataURLtoBlob(item.base64); window.open(URL.createObjectURL(blob), '_blank'); };
